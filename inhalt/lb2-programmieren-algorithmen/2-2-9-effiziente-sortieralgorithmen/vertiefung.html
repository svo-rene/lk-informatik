Aufgabe 2.2.9.6: Fehler finden – Elemente gehen verloren AFB III 
def merge(links, rechts):
    ergebnis = []
    i = j = 0
    while i < len(links) and j < len(rechts):
        if links[i] < rechts[j]:
            ergebnis.append(links[i])
            i = i + 1
        else:
            ergebnis.append(rechts[j])
            j = j + 1
    ergebnis.extend(links[i:])
    return ergebnis

def mergesort(liste):
    if len(liste) == 1:
        return liste
    mitte = len(liste) // 2
    return merge(mergesort(liste[:mitte]), mergesort(liste[mitte:]))
