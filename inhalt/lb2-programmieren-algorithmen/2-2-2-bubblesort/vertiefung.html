Aufgabe 2.2.2.5: Fehler finden – zwei fast richtige Bubblesorts AFB III 
# Variante A
def bubblesort_a(liste):
    n = len(liste)
    for durchlauf in range(n - 1):
        getauscht = False
        for j in range(n - 1 - durchlauf):
            if liste[j] > liste[j + 1]:
                temp = liste[j]
                liste[j] = liste[j + 1]
                liste[j + 1] = temp
                getauscht = True
            if not getauscht:
                break
    return liste

# Variante B
def bubblesort_b(liste):
    n = len(liste)
    for durchlauf in range(n - 1):
        for j in range(n - durchlauf):
            if liste[j] > liste[j + 1]:
                temp = liste[j]
                liste[j] = liste[j + 1]
                liste[j + 1] = temp
    return liste
