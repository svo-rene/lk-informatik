Ein Temperatursensor liefert gelegentlich ungültige negative Werte. Sie sollen aus der Liste entfernt werden:

def entferne_negative(werte):
    for i in range(len(werte)):
        if werte[i] < 0:
            werte.pop(i)

messung = [3, -1, -2, 5]
entferne_negative(messung)
print(messung)
