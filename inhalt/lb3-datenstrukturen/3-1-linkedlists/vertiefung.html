Eine Mitschülerin hat die Playlist um eine Methode zum Entfernen erweitert. Nutzer der App beschweren sich: „Manchmal wird ein neuer Titel angehängt, taucht aber nie auf.“ Hier ist ihre Klasse (Node wie in Beispiel 1):

class Playlist:
    def __init__(self):
        self._head = None
        self._tail = None
        self._anzahl = 0

    def anhaengen(self, titel):
        neu = Node(titel)
        if self._head is None:
            self._head = neu
        else:
            self._tail.next = neu
        self._tail = neu
        self._anzahl += 1

    def __len__(self):
        return self._anzahl

    def __str__(self):
        teile = []
        current = self._head
        while current is not None:
            teile.append(str(current.data))
            current = current.next
        return " -> ".join(teile)

    def entfernen(self, titel):
        current = self._head
        if current.data == titel:
            self._head = current.next
            self._anzahl -= 1
            return True
        while current.next is not None:
            if current.next.data == titel:
                current.next = current.next.next
                self._anzahl -= 1
                return True
            current = current.next
        return False
