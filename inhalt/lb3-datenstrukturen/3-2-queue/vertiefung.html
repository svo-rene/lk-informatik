Der IT-Support der Schule verwaltet Störungsmeldungen in einer Schlange. Ein Schüler hat dafür eine eigene Klasse geschrieben (Node wie in Beispiel 2) und zusätzlich einen Zähler eingebaut. Seitdem melden Lehrkräfte: „Mein Ticket wurde angenommen, aber die Statusanzeige sagt, es gibt keine offenen Tickets.“

class TicketQueue:
    def __init__(self):
        self._head = None
        self._tail = None
        self._anzahl = 0

    def is_empty(self):
        return self._anzahl == 0

    def enqueue(self, ticket):
        neu = Node(ticket)
        if self._tail is None:
            self._head = neu
        else:
            self._tail.next = neu
        self._tail = neu
        self._anzahl += 1

    def dequeue(self):
        if self.is_empty():
            raise IndexError("keine offenen Tickets")
        ticket = self._head.data
        self._head = self._head.next
        self._anzahl -= 1
        return ticket

    def peek(self):
        return self._head.data

def statusanzeige(t):                     # aus einem anderen Modul
    if t._head is not None:
        print("Nächstes Ticket:", t._head.data)
    else:
        print("Keine offenen Tickets")
