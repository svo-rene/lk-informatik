        self.farbe = farbe
        self.tagespreis = tagespreis
        self.gangzahl = gangzahl

        return f"Fahrrad: {self.rahmennummer}, {self.farbe}, {self.tagespreis} €/Tag, {self.gangzahl} Gänge"

    def mietpreis(self, tage):
        return self.tagespreis * tage
