Aufgabe 4.7.6: Fehler finden – Typangaben, die lügen AFB III

Eine Wetter-AG misst Temperaturen auf dem Schulhof. Das Programm hat Typangaben – trotzdem stimmt einiges nicht:

class Messreihe:
    def __init__(self, ort: str):
        self.ort: str = ort
        self.werte: list[float] = []

    def hinzufuegen(self, wert: float) -> None:
        self.werte.append(wert)

    def maximum(self) -> float:
        if not self.werte:
            return None
        return max(self.werte)

    def bericht(self) -> str:
        print(f"{self.ort}: max {self.maximum()} °C")


m = Messreihe("Schulhof")
m.hinzufuegen("21.5")
m.hinzufuegen(19.0)
text = m.bericht()
print(text.upper())
