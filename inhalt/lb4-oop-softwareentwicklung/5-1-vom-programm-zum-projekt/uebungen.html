Aufgabe 5.1.2: Wer importiert wen? – Pausenverkauf der Schülerfirma AFB II

Die Schülerfirma verkauft in der Hofpause Brezeln und Äpfel. Ein Mitglied hat das Programm schon auf vier Module verteilt:

# artikel.py
class Artikel:
    def __init__(self, name: str, preis_cent: int):
        self.name = name
        self.__preis_cent = preis_cent

    def get_preis(self) -> int:
        return self.__preis_cent
# lager.py
from artikel import Artikel

class Lager:
    def __init__(self):
        self.__artikel: dict[str, Artikel] = {}
        self.__bestand: dict[str, int] = {}
        self.__kasse_cent = 0

    def einlagern(self, artikel: Artikel, anzahl: int) -> None:
        self.__artikel[artikel.name] = artikel
        self.__bestand[artikel.name] = self.__bestand.get(artikel.name, 0) + anzahl

    def anzahl(self, name: str) -> int:
        return self.__bestand.get(name, 0)

    def get_kasse(self) -> int:
        return self.__kasse_cent

    def verkaufen(self, name: str) -> None:
        if self.anzahl(name) > 0:
            self.__bestand[name] -= 1
            self.__kasse_cent += self.__artikel[name].get_preis()
        else:
            print(f"{name} ist ausverkauft!")

    def nachbestellen(self) -> None:
        for name in self.__bestand:
            if self.__bestand[name] == 0:
                menge = int(input(f"Wie viele {name} nachbestellen? "))
                self.__bestand[name] = menge

l = Lager()
print("Lager-Test:", l.anzahl("Brezel"))
# anzeige.py
from lager import Lager

def zeige_bestand(lager: Lager, namen: list[str]) -> None:
    for name in namen:
        print(f"{name}: {lager.anzahl(name)} Stück")
    print(f"Kasse: {lager.get_kasse() / 100:.2f} €")
# main.py
from artikel import Artikel
from lager import Lager
from anzeige import zeige_bestand

def main() -> None:
    lager = Lager()
    lager.einlagern(Artikel("Brezel", 80), 2)
    lager.einlagern(Artikel("Apfel", 50), 1)
    lager.verkaufen("Apfel")
    lager.verkaufen("Apfel")
    lager.verkaufen("Brezel")
    zeige_bestand(lager, ["Brezel", "Apfel"])

if __name__ == "__main__":
    main()
